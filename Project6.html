# Ensure 'Date' column is in datetime format and sort the DataFrame
# (df is already loaded and cleaned from previous steps, but ensure Date is datetime)
df['Date'] = pd.to_datetime(df['Date'])
df = df.sort_values(by='Date')

# Identify columns representing regional average prices, excluding statewide
regional_price_columns = [col for col in df.columns if 'Average ($/gal)' in col and 'Statewide' not in col]

# Calculate the average price for each region
average_prices_by_region = df[regional_price_columns].mean().reset_index()
average_prices_by_region.columns = ['Region', 'Average Price ($/gal)']

# Extract just the region name for better labels
average_prices_by_region['Region'] = average_prices_by_region['Region'].str.replace(' Average ($/gal)', '')

# --- Visualization: Bar Chart for Average Heating Oil Prices by Region ---
plt.figure(figsize=(14, 7))
sns.barplot(x='Average Price ($/gal)', y='Region', data=average_prices_by_region, palette='viridis')
plt.title('Average Heating Oil Price by Region', fontsize=16)
plt.xlabel('Average Price ($/gal)', fontsize=12)
plt.ylabel('Region', fontsize=12)
plt.grid(axis='x', linestyle='--', alpha=0.7)
plt.tight_layout()
plt.show()
